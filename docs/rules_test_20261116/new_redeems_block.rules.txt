// 2026-11-16 抜け穴①（masa 🇦「ポイント交換はサーバーで確定する形にする」）: 交換の記録 kakiapp_redeems/{id} = { uid, pts, name, code, at }
//   作れるのは本人だけ。同じ書き込み（トランザクション）の中で、会員データの points がちょうど pts 減り、会員データの lastRedeem がこの id になっているときだけ
//   ＝「残高は減っていないのに交換の記録だけ作る」「1回の減算で記録を2つ作る」はできない。at はサーバー時刻。書き換え不可。読めるのは本人と運営、消せるのは運営だけ
//   アプリ（auth.js の kakiRedeem）は、この書き込みがサーバーに届いたのを確かめてから 4けたのコードを画面に出す
match /kakiapp_redeems/{id} {
  allow read: if request.auth != null && (isAdmin() || resource == null || resource.data.uid == request.auth.uid);
  allow create: if request.auth != null
    && request.resource.data.keys().hasOnly(["uid", "pts", "name", "code", "at"])
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.pts is int && request.resource.data.pts > 0
    && request.resource.data.name is string && request.resource.data.name.size() <= 60
    && request.resource.data.code is string && request.resource.data.code.matches("^[0-9]{4}$")
    && request.resource.data.at == request.time
    && exists(userDoc(request.auth.uid))
    && getAfter(userDoc(request.auth.uid)).data.lastRedeem == id
    && getAfter(userDoc(request.auth.uid)).data.points == get(userDoc(request.auth.uid)).data.points - request.resource.data.pts;
  allow update: if false;
  allow delete: if isAdmin();
}
