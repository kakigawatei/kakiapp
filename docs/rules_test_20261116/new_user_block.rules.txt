    //   ① 新規作成（resource==null）も「空のデータからの変更」として同じ検査を通す（前は中身を見ず +1000 まで・来店数も何件でも入った）
    //   ② キャンペーン受取 claimed: 足したキーが kakiapp_campaigns に実在・停止中でない・期間内（終了日の翌日まで猶予）・来店回数の条件を満たすときだけ。もらえるのはそのキャンペーンの points まで
    //   ③ 紹介した側 refClaimed: 足したキー（紹介された人の uid）ごとに kakiapp_referrals/{uid} の referrer が本人・status が qualified のときだけ（1件 50）
    //   ④ ガチャの加点は qrDate が今日（＝来店チェックインと同じ日）のときだけ／みんなの一杯の加点は今日の自分の投稿 kakiapp_board/{uid}_{今日} があるときだけ
    //   ⑤ teamVisitDay を今日にできるのは qrDate が今日のときだけ（学校の+1 は下の teamVisitOk）
    //   あわせて: ランクアップは段ごとの本来の額まで（常連100・猛者150・油神250・伝説350・米油500。前は全段 500）／来店ポイントは足した来店1件につき 10 まで（前は 20）
    //   上限の内訳: ルーレット10・ログイン1・ガチャ100（来店日だけ）・来店10×件数（未同期の過去日1件を含め2件まで）・ランク（段ごと）・キャンペーン（その額）・紹介50/件・紹介された50・投稿20
    //   学校対抗の報酬は admin が increment で加算（本人は書かない）。🟥 config.js の ranks[].minVisits・bonus・visitPoints・referral.points・board.points を上げるときは、ここも同じ日に直す
    //   🟥 1回の判定で使える式は 1000 まで（10/15 に来店の保存が全部拒否された原因）。前より軽くしてある: 新旧のデータ n／o・変わった項目 c・今日 t を userWriteOk で1回だけ作って渡す。
    //      めったに変わらない項目（ランク・キャンペーン・紹介・学校）の検査は、その項目が変わった書き込みのときだけ中に入る。直すときは docs/rules_test_20261116/margin.mjs で余裕を測り直す
    function isDay(v) { return v is string && v.matches("^[0-9]{4}-[0-9]{2}-[0-9]{2}$"); }
    function ymd(ts) { return string(ts.year()) + "-" + (ts.month() < 10 ? "0" : "") + string(ts.month()) + "-" + (ts.day() < 10 ? "0" : "") + string(ts.day()); }
    // 1日1回ものの日付（6項目）: 変えるなら「今日」だけ（消すのも不可）。式の数を減らすため、6項目とも「今日」の map と比べて1回で見る
    //   d.changedKeys()＝6項目のうち値が今日でないもの／d.removedKeys()＝6項目のうち新しいデータに無いもの → どちらも「変わった項目 c」に入っていてはいけない
    function uDays(c, d) { return !c.hasAny(d.changedKeys()) && !c.hasAny(d.removedKeys()); }
    // 受け取りの印（map）にキーがちょうど1つ増えた（古いキーは消えていない・値も変わっていない）
    function grew1(nm, om) { return nm is map && nm.size() == om.size() + 1 && nm.diff(om).affectedKeys().size() == 1; }
    // ランクアップ: 1回に1段・その段の来店数を満たすときだけ（知らないキーは通さない）
    function rankMin(k) { return {"silver": 5, "gold": 15, "rainbow": 40, "king": 100, "founder": 200}.get(k, 100000000); }
    function rankPts(k) { return {"silver": 100, "gold": 150, "rainbow": 250, "king": 350, "founder": 500}.get(k, 0); }
    function uRank(nr, orb, nv) { return nr == orb || (grew1(nr, orb) && nv >= rankMin(nr.keys().removeAll(orb.keys())[0])); }
    // キャンペーン: 足したキーが実在のキャンペーンで、今もらえる条件のときだけ（target の new/before は使い始めた日が自己申告のためここでは見ない）。実在しなければ get() が失敗＝拒否
    function campDoc(k) { return /databases/$(database)/documents/kakiapp_campaigns/$(k); }
    function campOk(d, t, nv) { return d.get("active", true) != false && (d.get("from", "") == "" || t >= d.from) && (d.get("until", "") == "" || ymd(request.time - duration.value(15, 'h')) <= d.until) && nv >= d.get("minVisits", 0); }
    function uClaim(nc, oc, t, nv) { return nc == oc || (grew1(nc, oc) && campOk(get(campDoc(nc.keys().removeAll(oc.keys())[0])).data, t, nv)); }
    // 紹介した側: 足したキーを1件ずつ紹介の記録と照合（最大10件）。読み取りは1件につき get() 1回だけ（1回の書き込みで読める数に上限があるため exists() と重ねない。記録が無ければ get() が失敗＝拒否）
    function refDoc(k) { return /databases/$(database)/documents/kakiapp_referrals/$(k); }
    function refIs(d) { return d.referrer == request.auth.uid && d.status == "qualified"; }
    function refAll(a, z) { return z >= 1 && z <= 10 && refIs(get(refDoc(a[0])).data) && (z < 2 || refIs(get(refDoc(a[1])).data)) && (z < 3 || refIs(get(refDoc(a[2])).data)) && (z < 4 || refIs(get(refDoc(a[3])).data)) && (z < 5 || refIs(get(refDoc(a[4])).data)) && (z < 6 || refIs(get(refDoc(a[5])).data)) && (z < 7 || refIs(get(refDoc(a[6])).data)) && (z < 8 || refIs(get(refDoc(a[7])).data)) && (z < 9 || refIs(get(refDoc(a[8])).data)) && (z < 10 || refIs(get(refDoc(a[9])).data)); }
    function refGrew(nm, om, a, z) { return nm.size() <= 10 && nm.size() == om.size() + z && nm.diff(om).affectedKeys().size() == z && refAll(a, z); }
    function uRefClaim(nm, om) { return nm == om || (nm is map && refGrew(nm, om, nm.keys().removeAll(om.keys()), nm.keys().removeAll(om.keys()).size())); }
    function uRefBack(n, o) { return o.get("referredBy", {}).get("status", "") != "qualified" || n.get("referredBy", {}).get("status", "") == "qualified"; }
    function uReferredNow(c, n, o) { return "referredBy" in c && n.get("referredBy", {}).get("status", "") == "qualified" && o.get("referredBy", {}).get("status", "") != "qualified" && get(/databases/$(database)/documents/kakiapp_referrals/$(request.auth.uid)).data.get("status", "") == "qualified"; }
    // visits を増やせるのは「qrDate が今日に変わる書き込み」1回だけ（ジェミ 10/14）。足せるのは 1〜2件・いずれも YYYY-MM-DD 形・最後の古い日付より後・末尾は今日。新規作成でも同じ
    //   a／b＝新旧の件数
    function uVisitsAdd(t, nv, ov, a, b, nq, oq) { return nv is list && nq == t && oq != t && nv[a - 1] == t && ((a == b + 1 && (b == 0 || nv[b] > ov[b - 1])) || (a == b + 2 && isDay(nv[b]) && (b == 0 || nv[b] > ov[b - 1]) && nv[b] < t)); }
    function uVisits(n, o, t) { return n.get("visits", []) == o.get("visits", []) || uVisitsAdd(t, n.get("visits", []), o.get("visits", []), n.get("visits", []).size(), o.get("visits", []).size(), n.get("qrDate", ""), o.get("qrDate", "")); }
    // めったに変わらない項目（ランク・キャンペーン・紹介・学校）。変わった項目だけ検査する
    function uRare(c, n, o, t) {
      return (!("rankBonus" in c) || uRank(n.get("rankBonus", {}), o.get("rankBonus", {}), n.get("visits", []).size()))
        && (!("claimed" in c) || uClaim(n.get("claimed", {}), o.get("claimed", {}), t, n.get("visits", []).size()))
        && (!("refClaimed" in c) || uRefClaim(n.get("refClaimed", {}), o.get("refClaimed", {})))
        && (!("referredBy" in c) || uRefBack(n, o))
        && (!("teamVisitDay" in c) || n.get("qrDate", "") == t)
        && uTeam(c) && uJoinDay(c);
    }
    // めったに変わらない項目の加点（uRare を通ったあとで呼ぶ）
    function rankGain(nr, orb) { return nr == orb ? 0 : rankPts(nr.keys().removeAll(orb.keys())[0]); }
    function campGain(nc, oc) { return nc == oc ? 0 : get(campDoc(nc.keys().removeAll(oc.keys())[0])).data.points; }
    function uRarePts(c, n, o) {
      return ("rankBonus" in c ? rankGain(n.get("rankBonus", {}), o.get("rankBonus", {})) : 0)
        + ("claimed" in c ? campGain(n.get("claimed", {}), o.get("claimed", {})) : 0)
        + ("refClaimed" in c ? 50 * (n.get("refClaimed", {}).size() - o.get("refClaimed", {}).size()) : 0)
        + (uReferredNow(c, n, o) ? 50 : 0);
    }
    function boardToday(t) { return exists(/databases/$(database)/documents/kakiapp_board/$(request.auth.uid + "_" + t)); }
    function uPoints(c, n, o, t, inc) {
      return inc <= 0 || inc <= ("rouletteDate" in c ? 10 : 0) + ("loginDate" in c ? 1 : 0)
        + (("gachaDate" in c && n.get("qrDate", "") == t) ? 100 : 0)
        + ("visits" in c ? 10 * (n.get("visits", []).size() - o.get("visits", []).size()) : 0)
        + (("boardDate" in c && boardToday(t)) ? 20 : 0)
        + (c.hasAny(["rankBonus", "claimed", "refClaimed", "referredBy"]) ? uRarePts(c, n, o) : 0);
    }
    function userWriteOk() {
      let n = request.resource.data;
      let o = resource == null ? {} : resource.data;
      let c = n.diff(o).affectedKeys();
      let t = ymd(request.time + duration.value(9, 'h'));
      return !c.hasAny(["awards", "qrRequired"])
        && uDays(c, n.diff({"rouletteDate": t, "gachaDate": t, "loginDate": t, "qrDate": t, "boardDate": t, "teamVisitDay": t}))
        && (!("visits" in c) || uVisits(n, o, t))
        && (!c.hasAny(["rankBonus", "claimed", "refClaimed", "referredBy", "teamId", "teamVisitDay"]) || uRare(c, n, o, t))
        && uPoints(c, n, o, t, n.get("points", 0) - o.get("points", 0))
        && uHours(c, t);
    }
